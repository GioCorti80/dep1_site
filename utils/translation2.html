CREATE TABLE `project_id.dataset_id.accodamenti_backup`
LIKE `project_id.dataset_id.accodamenti`;

INSERT INTO `project_id.dataset_id.accodamenti_backup`
(_PARTITIONTIME, colonna1, colonna2, colonna3)
SELECT
  _PARTITIONTIME,
  colonna1,
  colonna2,
  colonna3
FROM `project_id.dataset_id.accodamenti`
WHERE _PARTITIONTIME != TIMESTAMP('2026-11-18');
