Hi Subramiam, in the past, we used the adhoc-checks DAG to remove duplicates from a specific partition of the see_md_out table. The first step was to create a backup of the table, and the second step was to remove duplicates from the specified partition.

The tables containing the list of NDGs can only be modified through a release. Users do not have individual access to modify these tables.

However, I'm not sure whether it will be possible to request another file.

CREATE OR REPLACE PROCEDURE
  `my-project.snapshot.create_table_snapshots`(
    tables ARRAY,
    snapshot_date DATE
  )
BEGIN
  DECLARE table_name STRING;
  DECLARE snapshot_name STRING;
  DECLARE source_exists BOOL;

  FOR tbl IN (
    SELECT table_name
    FROM UNNEST(tables) AS table_name
  )
  DO
    SET table_name = tbl.table_name;

    -- Accetta solo nomi di tabelle semplici
    ASSERT REGEXP_CONTAINS(table_name, r'^[A-Za-z0-9_]+$')
      AS 'Nome tabella non valido';

    -- Verifica che la tabella sorgente esista
    SET source_exists = (
      SELECT COUNTIF(table_name = table_name_check) > 0
      FROM (
        SELECT table_name AS table_name_check
        FROM `my-project.tm_dataset.INFORMATION_SCHEMA.TABLES`
        WHERE table_type = 'BASE TABLE'
      )
      WHERE table_name_check = table_name
    );

    ASSERT source_exists AS 'Tabella sorgente inesistente';

    SET snapshot_name = FORMAT(
      '%s_%s',
      table_name,
      FORMAT_DATE('%Y%m%d', snapshot_date)
    );

    EXECUTE IMMEDIATE FORMAT("""
      CREATE SNAPSHOT TABLE IF NOT EXISTS
        `my-project.snapshot.%s`
      CLONE
        `my-project.tm_dataset.%s`
    """, snapshot_name, table_name);

  END FOR;
END;
