import bigframes.pandas as bpd
import pandas as pd
import re

def filter_and_extract_keywords(
    target_df: bpd.DataFrame,
    search_columns: list[str],
    keyword_bpd: bpd.DataFrame,
    kw_col: str = "keyword",
    type_col: str = "tipo",
    case_sensitive: bool = False
) -> bpd.DataFrame:
    
    # 1. Portiamo in RAM locale (Pandas) il dizionario delle keyword
    local_kw_df = keyword_bpd.to_pandas()
    local_kw_df[kw_col] = local_kw_df[kw_col].astype(str).str.strip()
    local_kw_df[type_col] = local_kw_df[type_col].astype(str).str.strip().str.lower()
    
    exact_list = local_kw_df[local_kw_df[type_col] == 'exactmatch'][kw_col].tolist()
    contains_list = local_kw_df[local_kw_df[type_col] == 'contains'][kw_col].tolist()
    
    exact_list = [k for k in exact_list if k]
    contains_list = [k for k in contains_list if k]
    
    if not exact_list and not contains_list:
        return target_df

    # --- FASE 1: FILTRO ---
    regex_contains_pattern = None
    if contains_list:
        regex_contains_pattern = "|".join([re.escape(k) for k in contains_list])

    final_mask = None

    for col in search_columns:
        col_mask = None
        
        if exact_list:
            if case_sensitive:
                mask_exact = target_df[col].isin(exact_list)
            else:
                mask_exact = target_df[col].str.lower().isin([k.lower() for k in exact_list])
            col_mask = mask_exact

        if contains_list:
            # Filtro contains (senza na=False, come da tua ottimizzazione)
            mask_contains = target_df[col].str.contains(regex_contains_pattern, case=case_sensitive, regex=True)
            col_mask = mask_contains if col_mask is None else (col_mask | mask_contains)

        final_mask = col_mask if final_mask is None else (final_mask | col_mask)

    # Creiamo il DataFrame filtrato su cui applicare le estrazioni pesanti
    filtered_df = target_df[final_mask].copy()


    # --- FASE 2: ESTRAZIONE ED ARRICCHIMENTO ---
    # Inizializziamo le nuove colonne (BigQuery le gestirà come NULL/String)
    filtered_df['keyword_found'] = None
    filtered_df['column_found'] = None

    # Prepariamo i pattern di estrazione con un singolo gruppo di cattura tra parentesi tonde ()
    # Il flag (?i) sposta il motore RE2 in modalità case-insensitive
    extract_exact_pat = None
    if exact_list:
        pat = "^(" + "|".join([re.escape(k) for k in exact_list]) + ")$"
        extract_exact_pat = pat if case_sensitive else "(?i)" + pat

    extract_contains_pat = None
    if contains_list:
        pat = "(" + "|".join([re.escape(k) for k in contains_list]) + ")"
        extract_contains_pat = pat if case_sensitive else "(?i)" + pat

    # Iteriamo per cercare la keyword e mappare la colonna (First Match Wins)
    for col in search_columns:
        
        if extract_exact_pat:
            # Estrae la stringa trovata. expand=False restituisce una singola Series
            ext = filtered_df[col].astype(str).str.extract(extract_exact_pat, expand=False)
            
            # Troviamo le righe in cui abbiamo avuto un match ORA e non lo avevamo già trovato in colonne precedenti
            update_mask = ext.notna() & filtered_df['keyword_found'].isna()
            
            # Assegnamento tramite .loc (BigFrames lo traduce in un CASE WHEN altamente efficiente in SQL)
            filtered_df.loc[update_mask, 'keyword_found'] = ext
            filtered_df.loc[update_mask, 'column_found'] = col

        if extract_contains_pat:
            ext = filtered_df[col].astype(str).str.extract(extract_contains_pat, expand=False)
            
            update_mask = ext.notna() & filtered_df['keyword_found'].isna()
            
            filtered_df.loc[update_mask, 'keyword_found'] = ext
            filtered_df.loc[update_mask, 'column_found'] = col

    return filtered_df
