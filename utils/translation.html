from datetime import date, timedelta
from calendar import monthrange


def get_date_range(periodo="trimestrale", mesi=None, riferimento=None):
    """
    Restituisce date_start e date_end del periodo precedente completo.

    Parametri:
        periodo: mensile, bimestrale, trimestrale,
                 semestrale, annuale
        mesi: numero di mesi precedenti completi (opzionale)
        riferimento: data di riferimento, utile per i test.
                     Se omessa, usa la data corrente.

    Returns:
        tuple[date, date]: (date_start, date_end)
    """

    oggi = riferimento or date.today()

    # Modalità con numero di mesi
    if mesi is not None:
        if not isinstance(mesi, int) or isinstance(mesi, bool) or mesi < 1:
            raise ValueError("Il parametro mesi deve essere un intero positivo.")

        # Primo giorno del mese corrente
        date_end = oggi.replace(day=1) - timedelta(days=1)

        # Primo giorno del mese iniziale
        indice_mese = date_end.year * 12 + date_end.month - 1
        indice_inizio = indice_mese - mesi + 1

        anno_start = indice_inizio // 12
        mese_start = indice_inizio % 12 + 1

        date_start = date(anno_start, mese_start, 1)

        return date_start, date_end

    # Periodi di calendario
    periodi = {
        "mensile": 1,
        "bimestrale": 2,
        "trimestrale": 3,
        "semestrale": 6,
        "annuale": 12,
    }

    periodo = periodo.lower().strip()

    if periodo not in periodi:
        raise ValueError(
            f"Periodo non valido: {periodo}. "
            f"Valori ammessi: {', '.join(periodi)}"
        )

    durata = periodi[periodo]

    # Indice del mese corrente, contando da zero
    indice_mese = oggi.year * 12 + oggi.month - 1

    # Inizio del periodo di calendario corrente
    inizio_periodo_corrente = (
        indice_mese // durata
    ) * durata

    # Ultimo mese del periodo precedente
    indice_fine = inizio_periodo_corrente - 1

    anno_end = indice_fine // 12
    mese_end = indice_fine % 12 + 1

    date_end = date(
        anno_end,
        mese_end,
        monthrange(anno_end, mese_end)[1]
    )

    # Primo mese del periodo precedente
    indice_inizio = inizio_periodo_corrente - durata

    anno_start = indice_inizio // 12
    mese_start = indice_inizio % 12 + 1

    date_start = date(anno_start, mese_start, 1)

    return date_start, date_end
